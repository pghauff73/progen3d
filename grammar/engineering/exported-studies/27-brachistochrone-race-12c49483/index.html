# Brachistochrone race. Race(terminalAngle dropMetres displaySlowdown railSegments).
# Try angles 165 or 210, a positive drop, and 24-64 rail segments.
# Beads slide from rest without friction. The hold after arrival is for display.
Start -> Race(210 1.85 2.4 44)

Race(angle drop slow n) ->
 [ Setup(radians(angle) drop/(1-cos(radians(angle))) drop slow n) ]
Setup(a r h slow n) ->
 [ Clockwork(a r h slow n r*(a-sin(a)) a*sqrt(r/9.80665)) ]
Clockwork(a r h slow n w tc) ->
 [ Exhibit(a r h n w tc sqrt(2*(w*w+h*h)/(9.80665*h))
           wrap(t/slow,0,sqrt(2*(w*w+h*h)/(9.80665*h))+0.36)) ]
