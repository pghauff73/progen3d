# Quasi-static flyball governor. Governor(lowRPM highRPM period lowerLength).
# Upper arms are 1.12 m. Suggested lower lengths are 1.12-1.32 m.
# Try 32-48 rpm over 6 s or 30-42 rpm over 5 s.
Start -> Governor(32 48 6 1.12)

Governor(lo hi period lower) ->
 [ Motion(lo+(hi-lo)*(1-cos(6.28318530718*t/period))/2 lower
          3*(lo+hi)*t-3*(hi-lo)*sin(6.28318530718*t/period)*period/6.28318530718) ]
Motion(rpm lower phase) ->
 [ Height(min(1.12,9.80665/(radians(6*rpm)*radians(6*rpm))) lower phase) ]
Height(h lower phase) ->
 [ Radius(h lower phase sqrt(1.12*1.12-h*h)) ]
