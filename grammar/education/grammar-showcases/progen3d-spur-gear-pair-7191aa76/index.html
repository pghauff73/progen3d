# ProGen3D | 20:40 external involute spur gears
# Open this file in the native grammar editor. Play or scrub 0 to 4 seconds.
# Start sets pinion rpm (signed). Each scene unit represents 10 mm.
# Module 2 mm, pressure angle 20 deg, face 8 mm, zero profile shift.
# Closed tooth sectors tile each wheel. Straight root extensions approximate
# the region below the base circle. The shaft angles are prescribed.
Start -> Drive(30)
Drive(rpm) -> Assembly(spin(t,rpm*6))
Assembly(inputAngle) ->
    [ T(-3 0 0) A(wrap(inputAngle,-180,180) 2) Pinion ]
    [ T(3 0 0) A(wrap(4.5-inputAngle/2,-180,180) 2) Wheel ]
